"""Serial DT comparisons at F=100 using the unchanged archived controllers."""

import argparse
import inspect
import json
from pathlib import Path

import run_case

HERE = Path(__file__).resolve().parent
campaign = run_case.existing


def completed(folder):
    for path in reversed(sorted(folder.glob("outcome_*.json"))):
        result = json.loads(path.read_text())
        if result.get("complete") and result.get("audit_passed"):
            return result
    return None


def save(folder, result):
    index = len(list(folder.glob("outcome_*.json"))) + 1
    result["factor"] = 100
    result["contact_conductance_mol_m2_s"] = 0.32
    (folder / f"outcome_{index:02d}.json").write_text(json.dumps(result, indent=2))
    print(json.dumps({"folder": str(folder), "result": result}), flush=True)
    if not result.get("complete") or not result.get("audit_passed"):
        raise SystemExit(1)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--case", choices=["all", "Db2", "Db6", "constant", "refinement"], default="all"
    )
    parser.add_argument("--resume-tip", type=Path)
    parser.add_argument("--birth-seed", type=Path)
    args = parser.parse_args()
    nominal = completed(HERE / "runs/dt_Db4_Fuller_F100_K032")
    assert nominal is not None, "Nominal case must finish before another numerical worker starts"
    source = inspect.getsource(campaign.dt)
    assert source.count('"1000"') == 2
    exec(
        compile(source.replace('"1000"', '"100"'), inspect.getsourcefile(campaign.dt), "exec"),
        campaign.__dict__,
    )
    for label, reference, constant in [
        ("Db2", 2e-10, False),
        ("Db6", 6e-10, False),
        ("constant", 4e-10, True),
    ]:
        if args.case not in ("all", label):
            continue
        folder = (
            HERE
            / "runs"
            / f"dt_Db{reference / 1e-10:g}_{'constant' if constant else 'Fuller'}_F100_K032"
        )
        folder.mkdir(parents=True, exist_ok=True)
        if completed(folder):
            print("Already complete and audited: " + label, flush=True)
            continue
        result = campaign.dt(
            folder, reference, constant, args.resume_tip, HERE / "replay_b.py", args.birth_seed
        )
        save(folder, result)
    if args.case not in ("all", "refinement"):
        return
    folder = HERE / "runs/dt_Db4_Fuller_F100_K032_C2p5"
    folder.mkdir(parents=True, exist_ok=True)
    if completed(folder):
        return
    tip = Path(nominal["b_tip"])
    out, result = campaign.invoke(
        folder,
        "c",
        campaign.HARNESS / "experiment_core2_radial_fast_wetting.py",
        [
            "--resume",
            tip.with_suffix(".accepted.pkl"),
            "--factor",
            "100",
            "--contact-conductance",
            ".32",
            "--dt",
            "2.5",
            "--until",
            "1766.5355008060146",
            "--steps",
            "1500",
            "--max-halvings",
            "12",
        ],
        4e-10,
    )
    if not result.get("bath_exit_reached"):
        save(folder, {"complete": False, "tip": str(out), "error": result.get("error")})
    audit, report = campaign.invoke(
        folder,
        "audit",
        campaign.HARNESS / "audit_finite_wetting_b.py",
        ["--result", tip, "--parts", *nominal["parts"], "--dry", out],
        4e-10,
    )
    save(
        folder,
        {
            "complete": True,
            "audit_passed": report["passed"],
            "tip": str(out),
            "b_tip": str(tip),
            "parts": nominal["parts"],
            "full_audit": str(audit),
            "final": result["final"],
            "scope": "Same accepted B history; regime-C maximum step 2.5 instead of 5 s",
        },
    )


if __name__ == "__main__":
    main()
