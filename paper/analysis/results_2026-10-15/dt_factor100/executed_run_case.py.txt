"""Use the archived property controller with only the prescribed factor changed."""

import argparse
import hashlib
import inspect
import json
from pathlib import Path
import sys

REPO = Path(__file__).resolve().parents[4]
HERE = Path(__file__).resolve().parent
SNAPSHOT = REPO / "paper/analysis/results_2026-10-14/temperature_properties/workspace_snapshot"
sys.path.insert(0, str(SNAPSHOT / "src"))
sys.path.insert(0, str(SNAPSHOT / "paper/analysis/harness"))

import run_temperature_property_campaign as existing  # noqa: E402


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--Db-ref", type=float, default=4e-10)
    parser.add_argument("--resume-tip", type=Path)
    parser.add_argument("--birth-seed", type=Path)
    args = parser.parse_args()
    folder = HERE / "runs" / f"dt_Db{args.Db_ref / 1e-10:g}_Fuller_F100_K032"
    folder.mkdir(parents=True, exist_ok=True)
    source = inspect.getsource(existing.dt)
    assert source.count('"1000"') == 2
    selected = source.replace('"1000"', '"100"')
    (HERE / "executed_dt_controller.py.txt").write_text(selected, encoding="utf-8")
    exec(compile(selected, inspect.getsourcefile(existing.dt), "exec"), existing.__dict__)
    result = existing.dt(
        folder,
        args.Db_ref,
        False,
        args.resume_tip,
        HERE / "replay_b.py",
        args.birth_seed,
    )
    result["factor"] = 100
    result["contact_conductance_mol_m2_s"] = 0.32
    result["controller_source_sha256"] = hashlib.sha256(
        Path(existing.__file__).read_bytes()
    ).hexdigest()
    result["scope"] = (
        "Original matching cp(T) feed and prescribed DT bath; only F1000 to F100; K=.32 unchanged"
    )
    index = len(list(folder.glob("outcome_*.json"))) + 1
    (folder / f"outcome_{index:02d}.json").write_text(
        json.dumps(result, indent=2), encoding="utf-8"
    )
    print(json.dumps(result), flush=True)
    if not result.get("complete") or not result.get("audit_passed"):
        raise SystemExit(1)


if __name__ == "__main__":
    main()
